  // Will be used for Default Project - All in the UI
  const buildAllHierarchy = () => {
    // Map<
    //   ProjectId,
    //   {
    //     project: ProjectData,
    //     parents: Map<
    //       ParentId,
    //       {
    //         parent: ParentData,
    //         atomics: AtomicData[]
    //       }
    //     >
    //   }
    // >
    const allHierarchy = new Map();

    // Extra lookup map so atomics can find parents directly by parentId
    const parentLookup = new Map();

    const projects = projectService.createProjectListSnapshot();
    const parents = parentService.createParentListSnapshot();
    const atomics = atomicService.createAtomicListSnapshot();

    // 1) Seed every project first so projects with 0 parents still show up
    for (const project of projects) {
      allHierarchy.set(project.id, {
        project,
        parents: new Map(),
      });
    }

    // 2) Add each parent into the correct project bucket
    for (const parent of parents) {
      const matchingProjectEntry = allHierarchy.get(parent.projectId);

      if (!matchingProjectEntry) {
        continue;
      }

      const parentEntry = {
        parent,
        atomics: [],
      };

      matchingProjectEntry.parents.set(parent.id, parentEntry);
      parentLookup.set(parent.id, parentEntry);
    }

    // 3) Add each atomic into the correct parent bucket
    for (const atomic of atomics) {
      const matchingParentEntry = parentLookup.get(atomic.parentId);

      if (!matchingParentEntry) {
        continue;
      }

      matchingParentEntry.atomics.push(atomic);
    }

    // 4) Build sidebar-friendly project count snapshot
    const projectCounts = [];
    let totalParentCount = 0;

    for (const [, projectEntry] of allHierarchy) {
      const parentCount = projectEntry.parents.size;

      projectCounts.push({
        id: projectEntry.project.id,
        name: projectEntry.project.name,
        parentCount,
      });

      if (projectEntry.project.name !== "All") {
        totalParentCount += parentCount;
      }
    }

    // 5) Overwrite "All" with the calculated total
    for (const projectSummary of projectCounts) {
      if (projectSummary.name === "All") {
        projectSummary.parentCount = totalParentCount;
        break;
      }
    }

    // Don't Delete - Study

    //     console.log("  ATOMICS:", parentEntry.atomics);
    //   }
    // }

    // Explicit Version - Don't Delete - Study
